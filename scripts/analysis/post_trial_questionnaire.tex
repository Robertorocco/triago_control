\section*{A. Workload (NASA Task Load Index\footnote{\footnotesize Hart \& Staveland (1988), \textit{NASA-TLX}; adapted here to a 7-point scale for rapid repeated administration between trials.})}

\scaleseven{1. Mental Demand -- How mentally demanding was the task?}{Very Low}{Very High}
\scaleseven{2. Physical Demand -- How physically demanding was the task?}{Very Low}{Very High}
\scaleseven{3. Temporal Demand -- How hurried or rushed was the pace of the task?}{Very Low}{Very High}
\scaleseven{4. Performance -- How successful were you in accomplishing the task?}{Perfect}{Failure}
\scaleseven{5. Effort -- How hard did you have to work to achieve your level of performance?}{Very Low}{Very High}
\scaleseven{6. Frustration -- How insecure, irritated, stressed, or annoyed were you?}{Very Low}{Very High}

\section*{B. Control, Trust \& Comfort\footnote{\footnotesize Items adapted from standard HRI trust/control constructs, e.g.\ Jian, Bisantz \& Drury (2000).}}

\scaleseven{7. I felt in control of the robot's motion.}{Strongly Disagree}{Strongly Agree}
\scaleseven{8. I trusted the robot to do what I intended.\ \footnotesize(leave blank if no assistance was active this trial)}{Strongly Disagree}{Strongly Agree}
\scaleseven{9. The robot's motion felt smooth and comfortable.}{Strongly Disagree}{Strongly Agree}
\scaleseven{10. I felt safe during this trial.}{Strongly Disagree}{Strongly Agree}
